<canvas id="gameCanvas" width="400" height="600"></canvas>

</div>

<script>

const canvas = document.getElementById("gameCanvas");

const ctx = canvas.getContext("2d");

// --- GAME SYSTEM SETTINGS ---

const TRACK_KEYS = ['d', 'f', 'j', 'k'];

const COLUMN_WIDTH = 400 / 4;

const TILE_HEIGHT = 130;

const HIT_LINE_Y = 600 - 120;

// States: "MENU", "BUFFER", "PLAYING", "GAME_OVER"

let currentState = "MENU";

let activeTiles = [];

let performancePoints = 500;

let hp = 100;

let currentRound = 1;

let tileSpeed = 6;

let baseTileSpeed = 7;

let currentDifficulty = "MEDIUM";

let score = 0;

let feedbackText = "";

let feedbackColor = "#000";

let feedbackTimer = 0;

let bufferStartTicks = 0;

let gameStartTicks = 0;

// --- AUDIO & ASSETS ENVIRONMENT ---

const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

const music = new Audio();

music.src = "song.mp3"; // Put your song file path here!

music.crossOrigin = "anonymous";

// Load Background Image

const bgImage = new Image();

bgImage.src = "background.jpg"; // Put your background file path here!

let hasBackground = false;

bgImage.onload = () => { hasBackground = true; };

// 🎵 MELODY TIMESTAMPS GENERATION (Simulating Librosa Onsets)

// This is a backup array. For perfect alignment, paste your actual array from Python here!

let originalMelodyTimes = [];

for (let t = 1.0; t < 300; t += 0.4 + Math.random() * 0.6) {

originalMelodyTimes.push(t);

}

let noteTimes = [];

// Synth Chime generator for 0ms tapping audio feedback latency

function playHitSound() {

const osc = audioCtx.createOscillator();

const gainNode = audioCtx.createGain();

osc.type = "sine";

osc.frequency.setValueAtTime(587.33, audioCtx.currentTime); // D5 Key

gainNode.gain.setValueAtTime(0.25, audioCtx.currentTime);

gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.08);

osc.connect(gainNode);

gainNode.connect(audioCtx.destination);

osc.start();

osc.stop(audioCtx.currentTime + 0.08);

}

// UI Buttons Boundaries

const buttons = {

EASY: { x: 100, y: 240, w: 200, h: 50, label: "EASY LOOP", color: "#28a745" },

MEDIUM: { x: 100, y: 310, w: 200, h: 50, label: "NORMAL LOOP", color: "#e68c0a" },

HARD: { x: 100, y: 380, w: 200, h: 50, label: "CHAOS LOOP", color: "#dc3545" },

STOP: { x: 400 - 90, y: 60, w: 75, h: 35, label: "STOP", color: "#dc3545" },

RESTART: { x: 120, y: 600 / 2 + 80, w: 160, h: 60, label: "MAIN MENU", color: "#3278f0" }

};

function getGrade(points) {

if (points >= 900) return "S";

if (points >= 750) return "A";

if (points >= 500) return "B";

if (points >= 300) return "C";

return "D";

}

function triggerFeedback(text, color, length = 30) {

feedbackText = text;

feedbackColor = color;

feedbackTimer = length;

}

function generateDifficultyNotes(mode) {

if (mode === "EASY") {

baseTileSpeed = 5;

return originalMelodyTimes.filter((_, i) => i % 3 === 0);

} else if (mode === "MEDIUM") {

baseTileSpeed = 7;

return originalMelodyTimes.filter((_, i) => i % 2 === 0);

} else {

baseTileSpeed = 10;

return [...originalMelodyTimes];

}

}

function startGame(mode) {

audioCtx.resume();

activeTiles = [];

performancePoints = 500;

hp = 100;

currentRound = 1;

score = 0;

feedbackText = "";

currentDifficulty = mode;

noteTimes = generateDifficultyNotes(mode);

tileSpeed = baseTileSpeed;

currentState = "BUFFER";

bufferStartTicks = Date.now();

}

function startNextRound() {

currentRound++;

activeTiles = [];

noteTimes = generateDifficultyNotes(currentDifficulty);

tileSpeed = baseTileSpeed + ((currentRound - 1) * 1.5);

triggerFeedback(`ROUND ${currentRound}!`, "#8232c8", 90);

currentState = "BUFFER";

bufferStartTicks = Date.now();

}

// --- CLICK INTERACTION INTERCEPTORS ---

window.addEventListener("mousedown", (e) => {

const rect = canvas.getBoundingClientRect();

const mx = e.clientX - rect.left;

const my = e.clientY - rect.top;

if (currentState === "MENU") {

if (mx >= buttons.EASY.x && mx <= buttons.EASY.x + buttons.EASY.w && my >= buttons.EASY.y && my <= buttons.EASY.y + buttons.EASY.h) startGame("EASY");

if (mx >= buttons.MEDIUM.x && mx <= buttons.MEDIUM.x + buttons.MEDIUM.w && my >= buttons.MEDIUM.y && my <= buttons.MEDIUM.y + buttons.MEDIUM.h) startGame("MEDIUM");

if (mx >= buttons.HARD.x && mx <= buttons.HARD.x + buttons.HARD.w && my >= buttons.HARD.y && my <= buttons.HARD.y + buttons.HARD.h) startGame("HARD");

} else if (currentState === "GAME_OVER") {

if (mx >= buttons.RESTART.x && mx <= buttons.RESTART.x + buttons.RESTART.w && my >= buttons.RESTART.y && my <= buttons.RESTART.y + buttons.RESTART.h) currentState = "MENU";

} else if (currentState === "PLAYING" || currentState === "BUFFER") {

if (mx >= buttons.STOP.x && mx <= buttons.STOP.x + buttons.STOP.w && my >= buttons.STOP.y && my <= buttons.STOP.y + buttons.STOP.h) {

music.pause();

music.currentTime = 0;

currentState = "MENU";

}

}

});

// --- KEYBOARD HIT REGISTERING ---

window.addEventListener("keydown", (e) => {

if (currentState !== "PLAYING") return;

const keyIndex = TRACK_KEYS.indexOf(e.key.toLowerCase());

if (keyIndex === -1) return;

let hitSuccessful = false;

for (let tile of activeTiles) {

if (tile.col === keyIndex && !tile.hit) {

let distance = Math.abs((tile.y + TILE_HEIGHT) - HIT_LINE_Y);

if (distance <= 35) {

tile.hit = true;

performancePoints = Math.min(1000, performancePoints + 40);

hp = Math.min(100, hp + 4);

score++;

triggerFeedback("PERFECT!", "#28a745");

playHitSound();

hitSuccessful = true;

break;

} else if (distance <= 75) {

tile.hit = true;

performancePoints = Math.min(1000, performancePoints + 15);

hp = Math.min(100, hp + 1);

score++;

triggerFeedback("GOOD", "#e68c0a");

playHitSound();

hitSuccessful = true;

break;

}

}

}

if (!hitSuccessful) {

performancePoints = Math.max(0, performancePoints - 50);

hp = Math.max(0, hp - 8);

triggerFeedback("MISS!", "#dc3545");

}

});

// --- MAIN ENGINE DRAWING LOOP ---

function update() {

// Background Drawing

if (hasBackground) {

ctx.drawImage(bgImage, 0, 0, 400, 600);

} else {

ctx.fillStyle = "#f5f5f5";

ctx.fillRect(0, 0, 400, 600);

}

if (currentState === "MENU") {

ctx.fillStyle = "rgba(255, 255, 255, 0.85)";

ctx.fillRect(20, 60, 360, 420);

ctx.fillStyle = "#282828";

ctx.font = "bold 32px Arial";

ctx.textAlign = "center";

ctx.fillText("Melody Tiles Endless", 200, 130);

// Draw Difficulty Buttons

['EASY', 'MEDIUM', 'HARD'].forEach(mode => {

let b = buttons[mode];

ctx.fillStyle = b.color;

ctx.beginPath(); ctx.roundRect(b.x, b.y, b.w, b.h, 8); ctx.fill();

ctx.fillStyle = "#fff";

ctx.font = "bold 20px Arial";

ctx.fillText(b.label, b.x + b.w/2, b.y + b.h/2 + 7);

});

} else if (currentState === "BUFFER") {

drawGuidelines();

let passed = (Date.now() - bufferStartTicks) / 1000;

let countdown = 3 - Math.floor(passed);

if (countdown <= 0) {

currentState = "PLAYING";

gameStartTicks = Date.now();

music.play().catch(err => console.log("Audio play deferred configuration required"));

} else {

ctx.fillStyle = "#3278f0";

ctx.font = "bold 40px Arial";

ctx.textAlign = "center";

ctx.fillText(countdown, 200, 270);

ctx.font = "16px Arial";

ctx.fillText(`ROUND ${currentRound}`, 200, 210);

}

} else if (currentState === "PLAYING") {

let elapsedTime = (Date.now() - gameStartTicks) / 1000;

// Auto-loop checking if track reaches end

if (music.ended && activeTiles.length === 0) {

startNextRound();

}

if (hp <= 0) {

currentState = "GAME_OVER";

music.pause();

music.currentTime = 0;

}

// Spawn Spanning Elements

if (noteTimes.length > 0 && elapsedTime >= noteTimes[0]) {

noteTimes.shift();

let randomCol = Math.floor(Math.random() * 4);

activeTiles.push({ x: randomCol * COLUMN_WIDTH, y: -TILE_HEIGHT, col: randomCol, hit: false });

}

drawGuidelines();

// Process Tiles Layout

for (let i = activeTiles.length - 1; i >= 0; i--) {

let tile = activeTiles[i];

tile.y += tileSpeed;

if (!tile.hit) {

ctx.fillStyle = "#28282d";

ctx.fillRect(tile.x, tile.y, COLUMN_WIDTH, TILE_HEIGHT);

}

if (tile.y > 600) {

if (!tile.hit) {

performancePoints = Math.max(0, performancePoints - 60);

hp = Math.max(0, hp - 12);

triggerFeedback("MISS!", "#dc3545");

}

activeTiles.splice(i, 1);

}

}

// HUD Interface Elements

ctx.fillStyle = "#c8c8c8";

ctx.beginPath(); ctx.roundRect(15, 15, 150, 15, 4); ctx.fill();

ctx.fillStyle = hp < 30 ? "#f03250" : "#28c764";

ctx.beginPath(); ctx.roundRect(15, 15, 150 * (hp / 100), 15, 4); ctx.fill();

ctx.fillStyle = "#282828";

ctx.font = "16px Arial";

ctx.textAlign = "left";

ctx.fillText(`Score: ${score} | Rnd: ${currentRound}`, 15, 50);

// Grade Badge

let grade = getGrade(performancePoints);

ctx.font = "bold 36px Arial";

ctx.textAlign = "right";

ctx.fillStyle = grade === "S" ? "#e6960a" : "#282828";

ctx.fillText(grade, 400 - 15, 45);

// Stop Button Box

let sb = buttons.STOP;

ctx.fillStyle = sb.color;

ctx.beginPath(); ctx.roundRect(sb.x, sb.y, sb.w, sb.h, 6); ctx.fill();

ctx.fillStyle = "#fff";

ctx.font = "bold 14px Arial";

ctx.textAlign = "center";

ctx.fillText(sb.label, sb.x + sb.w/2, sb.y + sb.h/2 + 5);

if (feedbackTimer > 0) {

ctx.fillStyle = feedbackColor;

ctx.font = "bold 24px Arial";

ctx.textAlign = "center";

ctx.fillText(feedbackText, 200, 350);

feedbackTimer--;

}

} else if (currentState === "GAME_OVER") {

ctx.fillStyle = "#dc3545";

ctx.font = "bold 40px Arial";

ctx.textAlign = "center";

ctx.fillText("SONG FAILED", 200, 220);

ctx.fillStyle = "#555";

ctx.font = "16px Arial";

ctx.fillText(`FAILED AT ROUND ${currentRound}`, 200, 260);

let finalGrade = getGrade(performancePoints);

ctx.fillStyle = finalGrade === "S" ? "#e6960a" : "#282828";

ctx.font = "bold 70px Impact";

ctx.fillText(finalGrade, 200, 340);

let rb = buttons.RESTART;

ctx.fillStyle = rb.color;

ctx.beginPath(); ctx.roundRect(rb.x, rb.y, rb.w, rb.h, 10); ctx.fill();

ctx.fillStyle = "#fff";

ctx.font = "bold 20px Arial";

ctx.fillText(rb.label, rb.x + rb.w/2, rb.y + rb.h/2 + 7);

}

requestAnimationFrame(update);

}

function drawGuidelines() {

ctx.strokeStyle = "#969696";

ctx.lineWidth = 2;

for (let i = 1; i < 4; i++) {

ctx.beginPath(); ctx.moveTo(i * COLUMN_WIDTH, 0); ctx.lineTo(i * COLUMN_WIDTH, 600); ctx.stroke();

}

ctx.strokeStyle = "#3296f0";

ctx.lineWidth = 4;

ctx.beginPath(); ctx.moveTo(0, HIT_LINE_Y); ctx.lineTo(400, HIT_LINE_Y); ctx.stroke();

}

requestAnimationFrame(update);

</script>

</body>
